cmake_minimum_required(VERSION 3.20)
project(DiscordPresenceManager VERSION 1.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(presence INTERFACE)
target_include_directories(presence INTERFACE src third_party)
if(WIN32)
    enable_language(RC)
    add_executable(DiscordPresenceManager WIN32 src/main.cpp assets/app.rc)
    target_include_directories(DiscordPresenceManager PRIVATE assets)
    target_compile_definitions(DiscordPresenceManager PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00 NOMINMAX)
    target_link_libraries(DiscordPresenceManager PRIVATE presence comdlg32 comctl32 shell32 ole32 gdi32 user32 d2d1 dwrite dwmapi)
    if(MINGW)
        target_link_options(DiscordPresenceManager PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
    if(MSVC)
        target_link_options(DiscordPresenceManager PRIVATE /MANIFEST:NO)
        target_compile_options(DiscordPresenceManager PRIVATE /utf-8 /W4)
        set_property(TARGET DiscordPresenceManager PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(presence_tests tests/presence_tests.cpp)
    target_link_libraries(presence_tests PRIVATE presence)
    if(MSVC)
        target_compile_options(presence_tests PRIVATE /utf-8)
    endif()
    add_test(NAME presence COMMAND presence_tests)
    add_executable(playlist_tests tests/playlist_tests.cpp)
    target_link_libraries(playlist_tests PRIVATE presence)
    if(MSVC)
        target_compile_options(playlist_tests PRIVATE /utf-8)
    endif()
    add_test(NAME playlist COMMAND playlist_tests)
    if(WIN32)
        add_executable(window_tests tests/window_tests.cpp assets/app.rc)
        target_include_directories(window_tests PRIVATE assets)
        target_compile_definitions(window_tests PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00 NOMINMAX)
        target_link_libraries(window_tests PRIVATE presence comdlg32 comctl32 shell32 ole32 gdi32 user32 d2d1 dwrite dwmapi)
        if(MSVC)
            target_compile_options(window_tests PRIVATE /utf-8)
            target_link_options(window_tests PRIVATE /MANIFEST:NO)
        endif()
        add_test(NAME window_lifecycle COMMAND window_tests)
        set_tests_properties(window_lifecycle PROPERTIES TIMEOUT 20)
        add_executable(ipc_tests tests/ipc_tests.cpp)
        target_link_libraries(ipc_tests PRIVATE presence)
        if(MSVC)
            target_compile_options(ipc_tests PRIVATE /utf-8)
        endif()
        add_test(NAME ipc COMMAND ipc_tests)
        set_tests_properties(ipc PROPERTIES TIMEOUT 60)
    endif()
endif()
